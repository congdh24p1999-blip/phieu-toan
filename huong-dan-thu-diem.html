Nhận điểm học sinh về Google Sheet

Làm một lần, khoảng 10 phút. Sau đó, mỗi khi học sinh nộp bài online, họ tên và điểm của các em tự ghi vào bảng tính của thầy cô. Muốn thì có thể nhận thêm email tổng hợp điểm mỗi tối.

Cần chuẩn bị

  • Máy tính (máy bàn hoặc máy xách tay). Bước cài đặt không làm được trên điện thoại.
  • Tài khoản Gmail đã đăng nhập trên trình duyệt (Chrome, Cốc Cốc, Edge… đều được).
  • Mẹo: mở trang hướng dẫn này trên điện thoại để vừa đọc vừa làm trên máy tính, hoặc in ra giấy.

Chưa làm bước nào.

  1. Tạo một bảng tính mới

    Bấm vào đây để tạo bảng tính mới

    Một thẻ mới mở ra, có bảng ô trống. Ở góc trên bên trái có dòng chữ Bảng tính không có tiêu đề: bấm vào đó, gõ tên, ví dụ Điểm lớp 9A, rồi nhấn phím Enter.

  2. Mở phần viết mã của bảng tính

    Trên hàng chữ phía trên bảng ô (Tệp, Chỉnh sửa, Xem, Chèn, Định dạng, Dữ liệu, Công cụ, Tiện ích, Trợ giúp), bấm Tiện ích, rồi bấm Apps Script (dòng có biểu tượng nhiều màu).

    Một thẻ mới mở ra. Giữa màn hình có khung chữ, dòng đầu là function myFunction() {.

    Một số máy ghi là Tiện ích mở rộng; máy dùng tiếng Anh: Extensions → Apps Script.

/* Ghi một kết quả vào trang tính của đề; trả về tên trang và có phải dòng mới không */
function ghiKetQua_(d) {
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const tab = String(d.sheet || 'Kết quả').replace(/[\[\]\*\?\/\\:]/g, ' ').slice(0, 90);
    const sh = ss.getSheetByName(tab) || ss.insertSheet(tab);
    if (sh.getLastRow() === 0) {
      sh.appendRow(HEAD);
      sh.setFrozenRows(1);
      sh.getRange(1, 1, 1, HEAD.length).setFontWeight('bold');
    }
    // chặn chuỗi bắt đầu bằng = + - @ để không bị hiểu thành công thức
    const safe = v => (typeof v === 'string' && /^[=+\-@]/.test(v)) ? "'" + v : v;
    const row = [new Date(), d.name, d.cls, d.score10, d.got, d.tot, d.tlSelf, d.nOK, d.n,
      d.minutes, d.autoSubmit ? 'có' : '', d.mode, d.code, d.parts, d.answers, String(d.id)]
      .map(v => safe(v == null ? '' : v));
    // cùng một lượt làm gửi lại (ví dụ sau khi tự chấm tự luận) thì ghi đè dòng cũ
    const n = sh.getLastRow() - 1;
    const hit = n > 0 ? sh.getRange(2, HEAD.length, n, 1).createTextFinder(String(d.id))
      .matchEntireCell(true).findNext() : null;
    if (hit) { sh.getRange(hit.getRow(), 1, 1, row.length).setValues([row]); return { tab, isNew: false }; }
    sh.appendRow(row);
    return { tab, isNew: true };
  } finally {
    lock.releaseLock();
  }
}

/* Học sinh nộp bài: trang làm bài gửi kết quả tới đây */
function doPost(e) {
  try {
    ghiKetQua_(JSON.parse(e.postData.contents));
  } catch (err) {
    return ContentService.createTextOutput('loi: ' + err);
  }
  return ContentService.createTextOutput('ok');
}

/* Mở địa chỉ trong trình duyệt để xem web app có chạy không.
   Trang hướng dẫn gọi kèm ?thu=1&cb=… để ghi một dòng thử và nhận lại kết quả. */
function doGet(e) {
  const p = (e && e.parameter) || {};
  const out = { ok: true, ban: 2 };
  if (p.thu === '1') {
    try {
      ghiKetQua_({ id: 'thu-' + Date.now(), sheet: 'Thử kết nối', name: 'Học sinh thử', cls: 'Thử', code: '000',
        mode: 'Thử kết nối', score10: 10, got: 10, tot: 10, answers: 'Dòng này do trang hướng dẫn gửi thử' });
      out.ghi = true;
      out.bang = SpreadsheetApp.getActiveSpreadsheet().getName();
    } catch (err) {
      out.ok = false;
      out.loi = String(err);
    }
  }
  if (p.cb && /^[A-Za-z_$][\w$]{0,40}$/.test(p.cb)) {
    return ContentService.createTextOutput(p.cb + '(' + JSON.stringify(out) + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput('Xưởng đề Toán: web app thu điểm đang chạy.');
}

/* Chạy hàm này một lần sau khi dán mã (và mỗi lần đổi cài đặt email):
   cấp quyền, gửi email thử, bật hoặc tắt lịch gửi email tổng hợp hằng ngày. */
function caiDat() {
  ScriptApp.getProjectTriggers()
    .filter(t => t.getHandlerFunction() === 'guiEmailTongHop')
    .forEach(t => ScriptApp.deleteTrigger(t));
  let lich = 'Không gửi email (chỉ ghi vào bảng tính).';
  if (EMAIL_KIEU === 'hang-ngay') {
    const gio = Math.min(23, Math.max(0, Math.floor(GIO_GUI)));
    ScriptApp.newTrigger('guiEmailTongHop').timeBased().everyDays(1).atHour(gio)
      .inTimezone(MUI_GIO).create();
    PropertiesService.getScriptProperties().setProperty('lanGuiTruoc', String(Date.now()));
    lich = 'Mỗi ngày khoảng ' + gio + ' giờ, một email tổng hợp các bài nộp mới (ngày không có bài nộp thì không gửi).';
  } else if (EMAIL_KIEU === 'moi-bai') {
    lich = 'Mỗi bài nộp mới một email.';
  }
  if (EMAIL_KIEU !== 'tat') {
    MailApp.sendEmail({
      to: nguoiNhan_(),
      subject: 'Xưởng đề Toán: đã cài đặt thu điểm',
      htmlBody: '<p>Bảng tính <b>' + esc_(SpreadsheetApp.getActiveSpreadsheet().getName()) +
        '</b> đã sẵn sàng nhận kết quả làm bài online.</p><p>Cách báo qua email: ' + esc_(lich) +
        '</p><p><a href="' + SpreadsheetApp.getActiveSpreadsheet().getUrl() + '">Mở bảng tính</a></p>'
    });
  }
  console.log('Đã cài đặt. ' + lich);
}

/* Gửi email tổng hợp các bài nộp từ lần gửi trước; lịch hằng ngày tự gọi hàm này. */
function guiEmailTongHop() {
  const props = PropertiesService.getScriptProperties();
  const tu = Number(props.getProperty('lanGuiTruoc')) || (Date.now() - 24 * 3600 * 1000);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const nhom = [];
  let tong = 0;
  ss.getSheets().forEach(sh => {
    if (sh.getLastRow() < 2) return;
    const v = sh.getDataRange().getValues();
    if (v[0][0] !== HEAD[0] || sh.getName() === 'Thử kết nối') return;   // bỏ qua trang thử
    const moi = v.slice(1).filter(r => r[0] instanceof Date && r[0].getTime() > tu);
    if (!moi.length) return;
    tong += moi.length;
    nhom.push({ ten: sh.getName(), url: ss.getUrl() + '#gid=' + sh.getSheetId(), rows: moi });
  });
  props.setProperty('lanGuiTruoc', String(Date.now()));
  if (!tong) return;
  const diem = r => (r[3] !== '' && r[3] != null) ? Number(r[3]) : (Number(r[5]) ? Number(r[4]) / Number(r[5]) * 10 : NaN);
  const so = x => (Math.round(x * 10) / 10).toString().replace('.', ',');
  let html = '<p>Từ ' + Utilities.formatDate(new Date(tu), MUI_GIO, 'HH:mm dd/MM') + ' đến nay có <b>' + tong +
    '</b> bài nộp mới (tính cả bài gửi lại sau khi tự chấm).</p>';
  nhom.forEach(g => {
    const ds = g.rows.map(diem).filter(x => !isNaN(x));
    const tb = ds.length ? ds.reduce((a, b) => a + b, 0) / ds.length : NaN;
    html += '<h3 style="margin:18px 0 4px">' + esc_(g.ten) + '</h3><p style="margin:0 0 6px">' + g.rows.length + ' bài' +
      (ds.length ? ' · điểm trung bình ' + so(tb) + ' · thấp nhất ' + so(Math.min.apply(null, ds)) +
        ' · cao nhất ' + so(Math.max.apply(null, ds)) + ' (thang 10)' : '') +
      ' · <a href="' + g.url + '">Xem trong bảng</a></p>';
    const rows = g.rows.slice().sort((a, b) => String(a[2]).localeCompare(String(b[2]), 'vi') || String(a[1]).localeCompare(String(b[1]), 'vi'));
    html += '<table cellpadding="4" style="border-collapse:collapse;font-size:14px"><tr style="background:#eef3f1">' +
      '<th align="left">Họ và tên</th><th align="left">Lớp</th><th align="right">Điểm</th><th align="left">Nộp lúc</th></tr>' +
      rows.slice(0, 60).map(r => '<tr style="border-top:1px solid #ddd"><td>' + esc_(r[1]) + '</td><td>' + esc_(r[2]) +
        '</td><td align="right">' + (isNaN(diem(r)) ? '' : so(diem(r))) + '</td><td>' +
        Utilities.formatDate(r[0], MUI_GIO, 'HH:mm dd/MM') + '</td></tr>').join('') + '</table>' +
      (rows.length > 60 ? '<p>… và ' + (rows.length - 60) + ' bài khác, xem đủ trong bảng tính.</p>' : '');
  });
  html += '<p style="color:#666;font-size:13px;margin-top:18px">Email tự động từ bảng tính <a href="' + ss.getUrl() + '">' +
    esc_(ss.getName()) + '</a> (Xưởng đề Toán). Muốn đổi giờ gửi hoặc tắt email: sửa phần CÀI ĐẶT EMAIL trong Apps Script rồi chạy lại hàm caiDat.</p>';
  MailApp.sendEmail({
    to: nguoiNhan_(),
    subject: 'Xưởng đề Toán: ' + tong + ' bài nộp mới (' + Utilities.formatDate(new Date(), MUI_GIO, 'dd/MM') + ')',
    htmlBody: html
  });
}

function guiEmailMotBai_(d, tab) {
  if (MailApp.getRemainingDailyQuota() < 1) return;   // hết lượt gửi trong ngày: kết quả vẫn có trong bảng
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const so = x => String(x).replace('.', ',');
  const diem = (d.score10 !== '' && d.score10 != null) ? so(d.score10) + '/10' : so(d.got) + '/' + so(d.tot) + ' điểm';
  MailApp.sendEmail({
    to: nguoiNhan_(),
    subject: 'Bài nộp: ' + String(d.name || 'Không tên').slice(0, 60) + (d.cls ? ' (' + String(d.cls).slice(0, 12) + ')' : '') + ' – ' + diem,
    htmlBody: '<p><b>' + esc_(d.name) + '</b>' + (d.cls ? ', lớp ' + esc_(d.cls) : '') + ' vừa nộp bài <b>' + esc_(tab) + '</b>' +
      (d.code ? ' (mã ' + esc_(d.code) + ')' : '') + '.</p><p>Điểm: <b>' + esc_(diem) + '</b> · Thời gian làm ' + esc_(d.minutes) +
      ' phút' + (d.autoSubmit ? ' · hết giờ, tự nộp' : '') + '</p><p>Điểm từng phần: ' + esc_(d.parts) + '</p>' +
      '<p><a href="' + ss.getUrl() + '">Mở bảng tính</a></p>'
  });
}

function nguoiNhan_() { return EMAIL_NHAN || Session.getEffectiveUser().getEmail(); }
function esc_(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c]);
}

  4. Bấm Chạy và cho phép

    Phía trên khung chữ có một ô chọn tên (cạnh nút Gỡ lỗi). Bấm vào ô đó, chọn caiDat. Sau đó bấm nút ▷ Chạy.

    Google sẽ hỏi quyền cho đoạn mã vừa dán. Bấm lần lượt:

    1. Xem lại quyền (Review permissions)
    2. chọn tài khoản Gmail của thầy cô
    3. chỉ với bản có email: hiện màn hình Google chưa xác minh ứng dụng này, đọc khung vàng ngay dưới đây
    4. nếu có ô Chọn tất cả thì đánh dấu ô đó
    5. Cho phép hoặc Tiếp tục (Allow / Continue)

    Bản có email: về màn hình “Google chưa xác minh ứng dụng này”

    Bản cơ bản không gặp màn hình này; bản có email thì có, vì mã xin thêm quyền gửi thư và hẹn giờ. Google hiện cảnh báo này với mọi đoạn mã tự viết chưa gửi Google duyệt, không phải vì mã có virus hay lừa đảo. Ở đây chính thầy cô vừa dán mã vào bảng tính của mình, mã chạy bằng tài khoản của thầy cô, và toàn bộ mã nằm ở bước 3 cho ai cũng đọc được. Mã chỉ ghi điểm vào đúng bảng tính này (bản có email thì thêm gửi thư về Gmail của thầy cô và hẹn giờ gửi thư); mã không đọc được Gmail hay tệp nào khác.

    Để đi tiếp: bấm chữ nhỏ Nâng cao (Advanced) ở góc dưới bên trái, rồi bấm Đi tới Dự án không có tiêu đề (không an toàn).

    Quy tắc an toàn: chỉ bấm Nâng cao khi chính thầy cô vừa dán mã từ trang Xưởng đề Toán này. Nếu ai đó gửi đường dẫn lạ rồi bảo bấm Nâng cao, đừng làm.

    Phía dưới khung chữ hiện Đã cài đặt…. Bản có email: vài giây sau, hộp thư nhận được thư Xưởng đề Toán: đã cài đặt thu điểm.

  5. Mở cửa để nhận điểm

    Góc trên bên phải, bấm nút xanh Triển khai, rồi bấm Tùy chọn triển khai mới.

    Trong hộp hiện ra:

    1. bấm hình bánh răng ⚙ cạnh chữ Chọn loại, chọn Ứng dụng web;
    2. ở ô Người có quyền truy cập, đổi thành đúng Bất kỳ ai. Không chọn Chỉ mình tôi hay Bất kỳ ai có Tài khoản Google: chọn hai ô đó thì học sinh không gửi điểm được.
    3. bấm Triển khai;
    4. nếu hiện hộp Cấp quyền truy cập (Authorize access): bấm vào và cho phép như bước 4. Hộp cho phép đôi khi mở thành cửa sổ nhỏ nằm khuất phía sau, hãy tìm trên thanh tác vụ.

    Tiếng Anh: Deploy → New deployment → ⚙ → Web app → Who has access: Anyone → Deploy.

  6. Chép địa chỉ và dán vào đây

    Hộp mới hiện dòng URL ứng dụng web (Web app URL), bên dưới có nút Sao chép. Bấm Sao chép, rồi bấm vào ô dưới đây và nhấn Ctrl + V.

    Không chép địa chỉ trên thanh địa chỉ của trình duyệt: đó là địa chỉ trang viết mã, không dùng được.

    Địa chỉ đúng có dạng https://script.google.com/macros/s/…/exec. Không thấy hộp này? Xem cách lấy lại địa chỉ.

    Bấm Lưu và gửi thử. Sau vài giây, ngay dưới nút hiện dòng chữ xanh Thành công…. Trong bảng tính, ở hàng thẻ dưới cùng (cạnh Trang tính1) có thêm thẻ Thử kết nối với dòng Học sinh thử. Thẻ thử này xoá đi cũng được. Nếu hiện chữ đỏ, làm theo cách sửa ghi ngay trong chữ đỏ đó.

Sau khi cài xong: giao bài và xem điểm

  1. Tạo mã QR mới để giao bài

    Ở trang tạo đề, chọn đề, bấm Cho học sinh làm online. Ô Thu điểm về Google Sheet phải có sẵn địa chỉ và bên dưới có dòng chữ xanh Đường dẫn bên dưới đã kèm địa chỉ thu điểm. Khi đó mới chép đường dẫn hoặc gửi mã QR.

    Đường dẫn, mã QR tạo trước khi cài không gửi điểm về. Phải tạo lại.

    Ô chỉ tự điền trên đúng máy tính, đúng trình duyệt đã làm hướng dẫn này. Dùng máy khác thì dán lại địa chỉ một lần. Lỡ làm mất địa chỉ: mở bảng tính → Tiện ích → Apps Script → Triển khai → Quản lý các bản triển khai.

  2. Làm thử một bài như học sinh

    Lấy điện thoại quét mã QR, gõ một tên thử, làm vài câu rồi nộp. Màn hình kết quả phải có dòng Đã gửi kết quả cho thầy cô, và trong bảng tính có dòng của bài vừa nộp. Được rồi mới gửi cho cả lớp.

    Khi có thu điểm, học sinh phải gõ họ tên trước khi làm. Dặn các em ghi đủ họ tên và lớp.

  3. Xem điểm ở đâu

    • Máy tính: vào sheets.google.com, bảng tính nằm đầu danh sách. Nên đánh dấu trang (Ctrl + D) để lần sau mở nhanh.
    • Điện thoại: cài ứng dụng Google Trang tính (Google Sheets), đăng nhập đúng Gmail đã cài.
    • Nhìn hàng thẻ dưới cùng của bảng tính: mỗi đề có một thẻ riêng, tên là tên đề và mã đề, ví dụ Đề kiểm tra giữa học kì I – mã 101. Kết quả không nằm ở thẻ Trang tính1.
    • Học sinh nộp xong vài giây là có dòng mới; bảng đang mở thì dòng mới tự hiện. Dùng bản có email thì mỗi tối khoảng 20 giờ còn có email tổng hợp.

  4. Đọc bảng điểm

    • Mỗi lần nộp là một dòng: thời điểm nộp, họ tên, lớp, điểm thang 10, điểm đạt, thời gian làm, điểm từng phần.
    • Cột Bài làm ghi lựa chọn từng câu, dấu ✓ là câu đúng. Ví dụ 1:B✓ 2:C 13:ĐSĐĐ✓ 19:2,5: câu 1 chọn B, đúng; câu 2 chọn C, sai; câu 13 chọn Đúng, Sai, Đúng, Đúng cho bốn ý, đúng hết; câu 19 trả lời 2,5, sai.
    • Muốn xếp theo điểm hay theo tên: bấm vào chữ cái đầu cột, rồi Dữ liệu → Sắp xếp trang tính.

  5. Bài có câu tự luận viết tay

    Trang không nhận được bài viết tay. Nên làm thế này:

    • Trong hộp Cho học sinh làm online, ở ô Câu tự luận chọn Giữ nguyên…, và tắt “Cho xem đáp án, lời giải sau khi nộp”. Học sinh làm tự luận ra giấy, điểm online chỉ tính phần máy chấm.
    • Học sinh nộp bài giấy, hoặc chụp ảnh gửi qua Zalo. Thầy cô chấm như bình thường.
    • Gõ điểm tự luận vào cột trống bên phải cột cuối cùng (cột Mã lượt làm), ví dụ đặt tên cột là Tự luận (GV chấm).
    • Nếu có cho xem lời giải, cột Tự luận (HS tự chấm) là điểm học sinh tự đánh dấu, chỉ để tham khảo.

    Đừng chèn cột vào giữa bảng, đừng đổi tên các thẻ đề: trang sẽ không ghi tiếp đúng chỗ. Chỉ thêm cột ở bên phải cột cuối.

Nếu gặp khó

Không thấy chữ “Tiện ích”.
Thầy cô đang mở trên điện thoại (ứng dụng điện thoại không có mục này), hoặc máy ghi là Tiện ích mở rộng, hoặc máy dùng tiếng Anh (tìm chữ Extensions). Hãy làm trên máy tính.
Ở bước 4 không thấy tên caiDat để chọn.
Mã chưa được lưu. Nhấn Ctrl + S, chờ một giây rồi chọn lại.
Bấm Chạy thì hiện chữ đỏ báo lỗi.
Thường do mã dán thiếu. Làm lại bước 3: bấm Chép mã, sang Apps Script nhấn Ctrl + A rồi Ctrl + V, lưu, rồi chạy lại.
Làm xong bước 5 mà không thấy địa chỉ (URL ứng dụng web).

Đừng chép địa chỉ trên thanh địa chỉ của trình duyệt: đó là địa chỉ trang viết mã. Làm như sau:

  1. Trong Apps Script, bấm nút xanh Triển khai → Quản lý các bản triển khai (Manage deployments).
  2. Nếu có một bản triển khai: bấm vào nó, bên phải có dòng Ứng dụng web và địa chỉ đuôi /exec, bấm Sao chép.
  3. Nếu danh sách trống, hoặc loại không phải Ứng dụng web: đóng hộp, làm lại bước 5, nhớ chọn bánh răng ⚙ → Ứng dụng web.
  4. Nếu sau khi bấm Triển khai hiện hộp Cấp quyền truy cập (Authorize access): bấm vào đó và cho phép như bước 4; xong mới hiện địa chỉ.
Bấm “Lưu và gửi thử” thì hiện chữ đỏ, hoặc bảng tính không có dòng mới.
Hay gặp nhất là ở bước 5 chưa đổi sang Bất kỳ ai (lưu ý: Bất kỳ ai có Tài khoản Google là sai, học sinh sẽ không gửi được). Sửa: trong Apps Script bấm Triển khai → Quản lý các bản triển khai → hình bút chì ✎ → ô Phiên bản chọn Phiên bản mới, ô Người có quyền truy cập chọn Bất kỳ ai → Triển khai. Địa chỉ giữ nguyên, bấm gửi thử lại.
Học sinh đã nộp mà bảng tính không có dòng nào.
  1. Xem đúng thẻ chưa: thẻ mang tên đề ở hàng dưới cùng, không phải Trang tính1. Nhiều thẻ thì bấm nút ☰ ở góc dưới bên trái để xem hết.
  2. Xem đúng Gmail chưa: bấm ảnh đại diện góc trên bên phải, đổi sang Gmail đã cài.
  3. Màn hình kết quả của học sinh có dòng Đã gửi kết quả cho thầy cô không? Không có, và lúc bắt đầu cũng không bắt gõ họ tên: mã QR được tạo khi chưa có địa chỉ thu điểm. Tạo lại mã QR như mục Sau khi cài xong ở trên.
Không nhận được email (bản có email).
Xem thư mục Thư rác (Spam) và Quảng cáo. Email tổng hợp gửi khoảng 20 giờ mỗi ngày, ngày nào không có bài nộp thì không gửi.
Muốn thôi nhận điểm.
Trong hộp Cho học sinh làm online, xoá trống ô Thu điểm về Google Sheet. Đường dẫn gửi học sinh sau đó sẽ không gửi điểm đi nữa.

Tên nút trên màn hình có thể khác đôi chút tùy phiên bản giao diện của Google. Nếu vẫn vướng, nhờ một đồng nghiệp rành máy tính làm cùng trên máy của thầy cô: chỉ cần làm một lần.

Bản có email: đổi cách nhận email

Xem cách đổi giờ gửi, nhận email từng bài, hoặc tắt email

Ở đầu mã (trong Apps Script) có ba dòng cài đặt:

  • EMAIL_KIEU: 'hang-ngay' là mỗi ngày một email tổng hợp (mặc định); 'moi-bai' là mỗi bài nộp một email; 'tat' là không gửi email.
  • GIO_GUI: giờ gửi email tổng hợp, từ 0 đến 23. Mặc định 20.
  • EMAIL_NHAN: để trống thì gửi về chính Gmail này; muốn gửi tới địa chỉ khác thì ghi vào giữa hai dấu nháy.

Sửa chữ trong dấu nháy, nhấn Ctrl + S, rồi chọn caiDat và bấm ▷ Chạy như bước 4. Riêng khi chuyển sang hoặc thôi kiểu 'moi-bai', làm thêm như mục “bảng tính không có dòng mới” ở trên (chọn Phiên bản mới rồi triển khai).

Kiểu mỗi bài một email tiện khi lớp ít học sinh. Gmail thường chỉ gửi được khoảng 100 email mỗi ngày; quá số đó thì thôi gửi email, điểm vẫn ghi đủ vào bảng tính.

Đang dùng bản cơ bản, muốn thêm email? Làm lại bước 3 (chọn bản có email) và bước 4, rồi trong Apps Script bấm Triển khai → Quản lý các bản triển khai → hình bút chì ✎ → Phiên bản mới → Triển khai. Địa chỉ nhận điểm giữ nguyên. Đổi ngược từ bản có email về bản cơ bản cũng làm y như vậy.

Lưu ý

Điểm đi thẳng từ máy học sinh vào bảng tính của thầy cô; Xưởng đề Toán không giữ bản nào.

Ai có đường dẫn bài làm cũng gửi được kết quả vào bảng, kể cả kết quả không thật, và đáp án được chấm ngay trên máy học sinh. Bảng điểm này hợp để theo dõi việc ôn luyện, chưa thay được bài kiểm tra có giám sát. Điểm phần tự luận do học sinh tự chấm, nên kiểm tra lại trên bài giấy.
